25def compute_crc16(data: bytes) -> int:
26 """
27 @brief Computes the CRC-16 (Cyclic Redundancy Check).
28 @details Uses the 0x1021 polynomial (CRC-CCITT).
29
30 @param data The binary data (bytes) to checksum.
31 @return int The calculated CRC as an unsigned 16-bit integer.
32 """
33 crc = 0x0000
34 poly = 0x1021
35 for b in data:
36 crc ^= (b << 8)
37 for _ in range(8):
38 if crc & 0x8000: crc = (crc << 1) ^ poly
39 else: crc <<= 1
40 crc &= 0xFFFF
41 return crc

127 def to_bytes(self) -> bytes:
128 """
129 @brief Converts the object into a byte sequence ready for network transmission.
130 @details Structure: [Start] [Len] [Dest] [Src] [Func] [Data] [CRC] [Stop].
131
132 @return bytes The complete binary frame.
133 """
134 # L = Dest(2) + Src(2) + Func(1) + Data(n)
135 l_val = 2 + 2 + 1 + len(self.data)
136
137 # Header (Little Endian)
138 header = struct.pack("<HHHB", l_val, self.dest, self.src, self.func)
139 body = header + self.data
140
141 # CRC (Big Endian >H on network !)
142 crc = compute_crc16(body)
143
144 return struct.pack("B", START_BYTE) + body + struct.pack(">H", crc) + struct.pack("B", STOP_BYTE)
