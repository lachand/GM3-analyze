28 def _crc16(self, data: bytes) -> int:
29 """ @brief Local CRC calculation. """
30 crc = 0x0000
31 poly = 0x1021
32 for b in data:
33 crc ^= (b << 8)
34 for _ in range(8):
35 if crc & 0x8000: crc = (crc << 1) ^ poly
36 else: crc <<= 1
37 crc &= 0xFFFF
38 return crc

40 def _slugify(self, text: str) -> str:
41 """
42 @brief Normalizes parameter names into valid JSON keys.
43 @param text Raw name (e.g., "T. Boiler [C]").
44 @return str Slug (e.g., "t_boiler_c").
45 """
46 text = text.lower()
47 text = re.sub(r'\[.*?\]', '', text)
48 text = re.sub(r'\s+', '_', text)
49 text = re.sub(r'[^a-z0-9_]', '', text)
50 return text.strip('_')
