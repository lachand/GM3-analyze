63 """
64 @brief Reads the incoming stream until a valid frame is obtained.
65 @details Implements a sliding buffer to assemble TCP fragments.

67

80 if not chunk:
81 return None

83
84 # Parsing
85 while True:
86 try:
87 start_idx = self._buffer.index(START_BYTE)
88 except ValueError:
89 self._buffer.clear()

91
92 # On aligne le buffer
93 if start_idx > 0:

95
96 # Header minimum : 68 L L
97 if len(self._buffer) < 3:
98 break

100 l_val = struct.unpack("<H", self._buffer[1:3])[0]
101 total_len = l_val + 6 # 68 + L(2) + Content(L) + CRC(2) + 16
102
103 if len(self._buffer) < total_len:
104 break # Trame incomplète, on attend
105
106 # Extraction
107 frame_bytes = self._buffer[:total_len]
108
109 # Validation CRC
110 # Body pour CRC = L(2) + Content(L) => indices 1 à 1+2+L
111 body_end = 1 + 2 + l_val
