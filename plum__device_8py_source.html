61 def _crc16(self, data: bytes) -> int:
62 """
63 @brief Internal CRC16 calculation.
64 @param data Binary data.
65 @return int Checksum.
66 """
67 crc = 0x0000
68 poly = 0x1021
69 for b in data:
70 crc ^= (b << 8)
71 for _ in range(8):
72 if crc & 0x8000: crc = (crc << 1) ^ poly
73 else: crc <<= 1
74 crc &= 0xFFFF
75 return crc

77 def _encode(self, value: Any, param_def: dict) -> bytes:
78 """
79 @brief Encodes a Python value into bytes based on parameter definition.
80 @param value The value to write (int, float, str).
81 @param param_def Dictionary describing the parameter (type, exponent).
82 @return bytes or None on error.
83 """
84 ptype = param_def['type']
85 exp = param_def['exponent']
86 if isinstance(value, (int, float)) and exp != 0:
87 value = int(round(value / (10 ** exp)))
88 try:
89 if ptype == "STRING": return str(value).encode('utf-8') + b'\x00'
90 elif ptype == "FLOAT": return struct.pack("<f", float(value))
91 elif ptype in ["BYTE", "SHORT_INT", "BOOL"]: return struct.pack("B", int(value))
92 elif ptype in ["INT", "WORD"]: return struct.pack("<h", int(value))
93 elif ptype in ["DWORD", "LONG_INT"]: return struct.pack("<i", int(value))
94 else: return None
95 except: return None

97 def _decode(self, data: bytes, param_def: dict) -> Any:
98 """
99 @brief Decodes a binary response into a Python value.
100 @param data Received bytes.
101 @param param_def Parameter definition.
102 @return Any Typed value (float, int, str) or None.
103 """
104 ptype = param_def['type']
105 exp = param_def['exponent']
106 try:
107 val = None
108 if ptype == "STRING":
109 if b'\x00' in data: val = data[:data.index(b'\x00')].decode('utf-8', 'ignore')
110 else: val = data.decode('utf-8', 'ignore')
111 elif ptype == "FLOAT" and len(data) >= 4:
112 val = struct.unpack("<f", data[:4])[0]
113 val = round(val, 2)
114 elif ptype in ["BYTE", "SHORT_INT", "BOOL"] and len(data) >= 1:
115 val = data[0]
116 elif ptype in ["INT", "WORD"] and len(data) >= 2:
117 val = struct.unpack("<h", data[:2])[0]
118 elif ptype in ["DWORD", "LONG_INT"] and len(data) >= 4:
119 val = struct.unpack("<i", data[:4])[0]
120
121 if val is not None and isinstance(val, (int, float)) and exp != 0:
122 val = val * (10 ** exp)
123 val = round(val, 2)
124 return val
125 except: return None

129 async def get_value(self, slug: str, retries: int = 5) -> Any:
130 """
131 @brief Reads a parameter with robust error handling (Async).
132 @details Wraps the synchronous `_sync_get_value` call in a thread.
133
134 @param slug Textual identifier of the parameter (e.g., "temp_boiler").
135 @param retries Max attempts before failure (default 5).
136 @return Any The read value or None if failed.
137 """
138 param = self.params_map.get(slug)
139 if not param: return None
140 pid = param['id']
141
142 for attempt in range(1, retries + 1):
143 val = await asyncio.to_thread(self._sync_get_value, pid, param)
144
145 if val is not None:
146 if attempt > 1:
147 logger.info(f"Finded'{slug}' with {attempt} tries.")
148 return val
149
150 if attempt < retries:
151 wait_time = 0.1 * attempt # Backoff progressif : 0.5s, 1.0s, 1.5s...
152 logger.warning(f"'{slug}' Timeout (try {attempt}/{retries}). Retry in {wait_time}s...")
153 await asyncio.sleep(wait_time)
154
155 logger.error(f"ABORTED '{slug}' after {retries} tries.")
156 return None

191 def _sync_get_value(self, pid: int, param: dict) -> Any:
192 """
193 @brief Blocking implementation of read operation (Executed in a thread).
194 @details Manages SessionID and frame construction.
195 @param pid Numeric ID of the parameter.
196 @param param Full definition for decoding.
197 @return Any Decoded value or None.
198 """
199 # Change session ID for each physical attempt
200 self.session_id = (self.session_id + 1) % 65000
201
202 payload = struct.pack("<HB BH", self.session_id, 1, 1, pid)
203 frame = self._build_frame(CMD_READ_VAL, payload)
204
205 resp = self._socket_transaction(frame, CMD_READ_VAL)
206
207 if resp and len(resp) > 2:
208 rec_sess = struct.unpack("<H", resp[0:2])[0]
209 if rec_sess == self.session_id or rec_sess == 0:
210 return self._decode(resp[7:], param)
211 return None

222 def _build_frame(self, cmd, payload):
223 """
224 @brief Builds a raw frame with Header and CRC.
225 """
226 l_val = 2 + 2 + 1 + len(payload)
227 header = struct.pack("<HHHB", l_val, DEST_ID, SOURCE_ID, cmd)
228 body = header + payload
229 chk = self._crc16(body)
230 return b'\x68' + body + struct.pack(">H", chk) + b'\x16'

232 def _socket_transaction(self, frame: bytes, expected_cmd: int) -> Optional[bytes]:
233 """
234 @brief Handles low-level TCP transaction: Connect -> Send -> Receive -> Close.
235 @details Uses a strict 2.0s timeout to prevent infinite blocking.
236
237 @param frame Binary frame to send.
238 @param expected_cmd Command expected in response (to filter stream).
239 @return bytes Response payload or None if timeout/error.
240 """
241 sock = None
242 try:
243 sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
244 sock.settimeout(2.0)
245
246 sock.connect((self.ip, self.port))
247 sock.send(frame)
248
249 buffer = bytearray()
250 start_time = time.time()
251
252 while time.time() - start_time < 2.0:
253 try:
254 chunk = sock.recv(2048)
255 if not chunk: break
256 buffer.extend(chunk)
257
258 if b'\x68' in buffer:
259 idx = buffer.find(b'\x68')
260 if idx != -1 and len(buffer) > idx + 8:
261 cmd_rec = buffer[idx+7]
262 if cmd_rec == (expected_cmd | 0x80):
263 return buffer[idx+8 : -3]

265 except socket.timeout:

270 except Exception:

274 try:
